"risk": request.get("risk", "medium"),

"status": "pending_review",

  
  
  
  
  
  
  
  
  
  
  
  
  
  
  
  
  
  
  


"created_at": datetime.utcnow().isoformat(),

  
  
  
  
  
  
  
  
  
  
  
  
  
  
  
  
  
  
  


"approvals": [],
